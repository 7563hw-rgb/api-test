// ESP32와 동일한 UUID const SERVICE_UUID = '4fafc201-1fb5-459e-8fcc-c5c9c331914b'; const CHARACTERISTIC_UUID = 'beb5483e-36e1-4688-b7f5-ea07361b26a8'; let myCharacteristic; // 사용자 버튼 클릭 시 호출되는 연결 함수 (보안상 반드시 버튼 클릭으로 실행되어야 함) async function connectBluetooth() { try { console.log('블루투스 장치 검색 중...'); const device = await navigator.bluetooth.requestDevice({ filters: [{ name: 'Haptic_Belt' }], optionalServices: [SERVICE_UUID] }); const server = await device.gatt.connect(); const service = await server.getPrimaryService(SERVICE_UUID); myCharacteristic = await service.getCharacteristic(CHARACTERISTIC_UUID); console.log('햅틱 복대 연결 완료!'); } catch (error) { console.error('블루투스 연결 실패:', error); } } // Tmap에서 다음 웨이포인트까지의 각도(Target Angle)가 계산될 때마다 호출하는 함수 async function sendTargetAngleToBelt(angle) { if (!myCharacteristic) return; try { // 각도를 문자열로 변환하여 전송 (예: "90") const encoder = new TextEncoder(); await myCharacteristic.writeValue(encoder.encode(angle.toString())); } catch (error) { console.error('데이터 전송 실패:', error); } }
